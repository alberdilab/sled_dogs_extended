sample_metadata <- sample_metadata %>%
  mutate(country = if_else(country == "Shangai domestic dog", "Shangai", country)) %>% 
  mutate(
    dog_type = case_when(
      breed == "Wolf"        ~ "wolf",
      country == "Greenland" ~ "sled_dog",
      TRUE                   ~ "domestic"

  )
