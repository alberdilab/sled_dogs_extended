Chapter 12 temporal divergence

Microbiome divergence from baseline increased significantly over time (day effect: p < 0.001)

meta <- sample_metadata_ordered %>%
  rownames_to_column("sample")

baseline_df <- meta %>%
  filter(day == "day_1") %>%
  select(animal_id, baseline_sample = sample)

df_dist <- meta %>%
  left_join(baseline_df, by = "animal_id") %>%
  mutate(dist = as.matrix(neutral_dis)[cbind(sample, baseline_sample)]) %>%
  mutate(
    day_num = as.numeric(gsub("day_", "", day)),
    animal_id = factor(animal_id)
  )

m1 <- lmerTest::lmer(dist ~ day_num + (1 | animal_id), data = df_dist) #Do individuals diverge from their baseline over time? Does distance from baseline (day 1) change over time, while accounting for repeated measures within each animal?
